/*
  MUY IMPORTANTE:

  "basico" en el catálogo NO significa que Chef Luna
  pueda asumir que el usuario lo tiene.

  Ejemplos que NO deben asumirse automáticamente:

  - leche
  - café
  - huevo
  - aceite
  - azúcar
  - mantequilla
  - tortilla
  - cebolla
  - ajo

  Aunque alguno esté clasificado como "básico",
  debe aparecer en la despensa del usuario.

  Solamente dejamos implícitos ingredientes realmente
  universales cuya ausencia no debería bloquear
  una recomendación normal.
*/

const INGREDIENTES_ASUMIBLES =
  new Set<string>([
    'agua',
    'sal',
  ]);


// ======================================================
// 🧠 ¿SE PUEDE ASUMIR?
// ======================================================

function esIngredienteAsumible(
  id: string
): boolean {

  return INGREDIENTES_ASUMIBLES.has(
    id
  );

}


// ======================================================
// 📊 ORGANIZAR RESULTADOS
// ======================================================

export function organizarResultados(
  resultados: ResultadoBusqueda[]
) {

  // ====================================================
  // ✅ LISTAS
  // ====================================================

  const recetasListas =
    resultados
      .filter(
        receta =>
          receta.puedeCocinar &&
          receta.dentroPresupuesto
      )
      .sort(
        (a, b) => {

          // =============================================
          // 1. MAYOR COMPATIBILIDAD
          // =============================================

          if (
            b.compatibilidad !==
            a.compatibilidad
          ) {

            return (
              b.compatibilidad -
              a.compatibilidad
            );

          }


          // =============================================
          // 2. MÁS RÁPIDA
          // =============================================

          if (
            a.tiempoMinutos !==
            b.tiempoMinutos
          ) {

            return (
              a.tiempoMinutos -
              b.tiempoMinutos
            );

          }


          // =============================================
          // 3. MENOR COSTO COMPLETO
          // =============================================

          return (
            a.costoTotal -
            b.costoTotal
          );

        }
      );


  // ====================================================
  // 🟠 PARCIALES
  // ====================================================

  const recetasParciales =
    resultados
      .filter(
        receta =>
          !receta.puedeCocinar ||
          !receta.dentroPresupuesto
      )
      .sort(
        (a, b) => {

          // =============================================
          // 1. DENTRO DEL PRESUPUESTO
          // =============================================

          if (
            a.dentroPresupuesto !==
            b.dentroPresupuesto
          ) {

            return a.dentroPresupuesto
              ? -1
              : 1;

          }


          // =============================================
          // 2. MENOS FALTANTES
          // =============================================

          if (
            a.faltantes.length !==
            b.faltantes.length
          ) {

            return (
              a.faltantes.length -
              b.faltantes.length
            );

          }


          // =============================================
          // 3. MENOR GASTO PENDIENTE
          // =============================================

          if (
            a.costoFaltantes !==
            b.costoFaltantes
          ) {

            return (
              a.costoFaltantes -
              b.costoFaltantes
            );

          }


          // =============================================
          // 4. MAYOR COMPATIBILIDAD
          // =============================================

          if (
            b.compatibilidad !==
            a.compatibilidad
          ) {

            return (
              b.compatibilidad -
              a.compatibilidad
            );

          }


          // =============================================
          // 5. MÁS RÁPIDA
          // =============================================

          if (
            a.tiempoMinutos !==
            b.tiempoMinutos
          ) {

            return (
              a.tiempoMinutos -
              b.tiempoMinutos
            );

          }


          return (
            a.costoTotal -
            b.costoTotal
          );

        }
      );


  // ====================================================
  // 🏆 MEJOR RESULTADO BASE
  // ====================================================

  /*
    Esto es solamente el orden inicial.

    ChefEngine posteriormente toma la decisión final
    dependiendo de la crisis seleccionada.
  */

  const mejorReceta =
    recetasListas[0] ??
    recetasParciales[0] ??
    null;


  return {

    mejorReceta,

    recetasListas,

    recetasParciales,

  };

}


// ======================================================
// 🧺 NORMALIZAR INGREDIENTES DEL USUARIO
// ======================================================

function normalizarIngredientesUsuario(
  ingredientesUsuario: string[]
): string[] {

  const ids =
    ingredientesUsuario
      .map(
        ingrediente =>
          resolverIngredienteId(
            ingrediente
          )
      )
      .filter(
        (
          id
        ): id is string =>
          id !== null
      );


  return [
    ...new Set(
      ids
    ),
  ];

}


// ======================================================
// 🍳 INGREDIENTES QUE REALMENTE BLOQUEAN
// ======================================================

function obtenerIngredientesNecesarios(
  receta: RecetaConCosto
): string[] {

  const canonicos =
    receta.ingredientesClave
      .map(
        id => {

          const resuelto =
            resolverIngredienteId(
              id
            );


          if (
            !resuelto
          ) {

            if (
              __DEV__
            ) {

              console.warn(
                `⚠️ Ingrediente inválido "${id}" en receta "${receta.titulo}"`
              );

            }


            return null;

          }


          return resuelto;

        }
      )
      .filter(
        (
          id
        ): id is string =>
          id !== null
      );


  // ====================================================
  // 🧹 EVITAR DUPLICADOS
  // ====================================================

  const unicos =
    [
      ...new Set(
        canonicos
      ),
    ];


  // ====================================================
  // 🧂 QUITAR ÚNICAMENTE INGREDIENTES ASUMIBLES
  // ====================================================

  /*
    ANTES:

      !ingrediente?.basico

    Eso hacía desaparecer ingredientes reales de la
    evaluación.

    AHORA:

    Solo agua y sal pueden omitirse como requisito
    explícito.

    Todo lo demás debe existir en la despensa.
  */

  return unicos.filter(
    id =>
      !esIngredienteAsumible(
        id
      )
  );

}


// ======================================================
// 💰 CALCULAR LO QUE FALTA COMPRAR
// ======================================================

function calcularCostoFaltantes(
  receta: RecetaConCosto,
  faltantes: string[],
  regionId: RegionId
): number {

  if (
    faltantes.length === 0
  ) {

    return 0;

  }


  const idsFaltantes =
    new Set(
      faltantes
    );


  const costo =
    receta.ingredientes.reduce(
      (
        total,
        ingrediente
      ) => {

        const idCanonico =
          resolverIngredienteId(
            ingrediente.id
          );


        if (
          !idCanonico ||
          !idsFaltantes.has(
            idCanonico
          )
        ) {

          return total;

        }


        return (
          total +
          calcularCostoIngrediente(
            {
              id:
                idCanonico,

              cantidad:
                ingrediente.cantidad,

              unidad:
                ingrediente.unidad,
            },
            regionId
          )
        );

      },
      0
    );


  return (
    Math.round(
      costo *
      100
    ) /
    100
  );

}

// 🔎 BUSCAR RECETAS
// ======================================================

export function evaluarReceta(
  receta: RecetaConCosto,
  ingredientesUsuario: string[],
  presupuesto: number,
  regionId: RegionId = REGION_NACIONAL_ID
): ResultadoBusqueda | null {
  const ingredientesDisponibles = new Set(normalizarIngredientesUsuario(ingredientesUsuario));
  const presupuestoSeguro = Number.isFinite(presupuesto) ? Math.max(0, presupuesto) : 0;

    if (!receta.ingredientesClave ||
        receta.ingredientesClave.length === 0) {
        if (__DEV__) {
            console.warn('⚠️ Receta sin ingredientesClave:', receta.titulo);
        }
        return null;
    }
    const ingredientesNecesarios = obtenerIngredientesNecesarios(receta);
    const faltantes = ingredientesNecesarios.filter(id => !ingredientesDisponibles.has(id));
    const coincidencias = ingredientesNecesarios.length -
        faltantes.length;
    const compatibilidad = ingredientesNecesarios.length === 0
        ? 100
        : Math.round((coincidencias /
            ingredientesNecesarios.length) *
            100);
    let estadoReceta: 'lista' | 'casi' | 'faltante';
    if (faltantes.length === 0) {
        estadoReceta =
            'lista';
    }
    else if (faltantes.length <= 2) {
        estadoReceta =
            'casi';
    }
    else {
        estadoReceta =
            'faltante';
    }
    const puedeCocinar = faltantes.length === 0;
    const costoFaltantes = calcularCostoFaltantes(receta, faltantes, regionId);
    const dentroPresupuesto = costoFaltantes <=
        presupuestoSeguro;
    const razones: string[] = [];
    if (puedeCocinar) {
        razones.push('Puedes cocinarla con lo que realmente tienes en tu despensa.');
    }
    else if (faltantes.length === 1) {
        razones.push('Solo te falta 1 ingrediente.');
    }
    else if (faltantes.length === 2) {
        razones.push('Solo te faltan 2 ingredientes.');
    }
    else {
        razones.push(`Te faltan ${faltantes.length} ingredientes.`);
    }
    if (puedeCocinar) {
        razones.push('No necesitas comprar nada para completar esta receta.');
    }
    else if (dentroPresupuesto) {
        razones.push(`La compra pendiente cuesta aproximadamente $${Math.round(costoFaltantes)} y entra en tu presupuesto.`);
    }
    else {
        razones.push(`Necesitarías aproximadamente $${Math.round(costoFaltantes)} para completar esta receta.`);
    }
    if (receta.tiempoMinutos <= 15) {
        razones.push('Es una preparación muy rápida.');
    }
    else if (receta.tiempoMinutos <= 30) {
        razones.push('Tiene un tiempo de preparación razonable.');
    }
    if (receta.costoTotal <= 40) {
        razones.push('El valor completo de la receta es bajo.');
    }
    return {
        ...receta,
        estadoReceta,
        coincidencias,
        totalIngredientes: ingredientesNecesarios.length,
        compatibilidad,
        faltantes,
        costoFaltantes,
        puedeCocinar,
        dentroPresupuesto,
        razones,
    } satisfies ResultadoBusqueda;

}

export function buscarRecetas(
  ingredientesUsuario: string[],
  presupuesto: number,
  recetasConCosto: RecetaConCosto[],
  regionId: RegionId =
    REGION_NACIONAL_ID
): ResultadoBusqueda[] {

  // ====================================================
  // 🧺 DESPENSA
  // ====================================================

  const ingredientesUnicos =
    normalizarIngredientesUsuario(
      ingredientesUsuario
    );


  // ====================================================
  // 🍳 ANALIZAR RECETAS
  // ====================================================

  const resultados =
    recetasConCosto
      .map(
        receta => evaluarReceta(receta, ingredientesUsuario, presupuesto, regionId)
      )
      .filter(
        (
          receta
        ): receta is ResultadoBusqueda =>
          receta !== null
      );


  // ====================================================
  // 🧹 FILTRAR RESULTADOS MUY LEJANOS
  // ====================================================

  const resultadosUtiles =
    resultados.filter(
      receta => {

        /*
          Sin despensa registrada se conserva el
          comportamiento exploratorio.
        */

        if (
          ingredientesUnicos.length === 0
        ) {

          return true;

        }


        /*
          Una receta compuesta únicamente por ingredientes
          asumibles, por ejemplo agua/sal, sigue siendo válida.
        */

        if (
          receta.totalIngredientes === 0
        ) {

          return true;

        }


        /*
          Debe compartir al menos un ingrediente real
          con la despensa.
        */

        if (
          receta.coincidencias > 0
        ) {

          return true;

        }


        /*
          También conservamos oportunidades inmediatas:

          0 coincidencias
          + exactamente 1 ingrediente faltante.

          Chef Luna puede sugerir esa compra cuando tenga
          sentido.
        */

        return (
          receta.faltantes.length === 1
        );

      }
    );


  // ====================================================
  // 🏆 ORDEN BASE
  // ====================================================

  return resultadosUtiles.sort(
    (a, b) => {

      // ===============================================
      // 1. YA PUEDES COCINARLA
      // ===============================================

      if (
        a.puedeCocinar !==
        b.puedeCocinar
      ) {

        return a.puedeCocinar
          ? -1
          : 1;

      }


      // ===============================================
      // 2. DENTRO DEL PRESUPUESTO
      // ===============================================

      if (
        a.dentroPresupuesto !==
        b.dentroPresupuesto
      ) {

        return a.dentroPresupuesto
          ? -1
          : 1;

      }


      // ===============================================
      // 3. MENOS FALTANTES
      // ===============================================

      if (
        a.faltantes.length !==
        b.faltantes.length
      ) {

        return (
          a.faltantes.length -
          b.faltantes.length
        );

      }


      // ===============================================
      // 4. MENOR GASTO PENDIENTE
      // ===============================================

      if (
        a.costoFaltantes !==
        b.costoFaltantes
      ) {

        return (
          a.costoFaltantes -
          b.costoFaltantes
        );

      }


      // ===============================================
      // 5. MAYOR COMPATIBILIDAD
      // ===============================================

      if (
        b.compatibilidad !==
        a.compatibilidad
      ) {

        return (
          b.compatibilidad -
          a.compatibilidad
        );

      }


      // ===============================================
      // 6. MÁS RÁPIDA
      // ===============================================

      if (
        a.tiempoMinutos !==
        b.tiempoMinutos
      ) {

        return (
          a.tiempoMinutos -
          b.tiempoMinutos
        );

      }


      // ===============================================
      // 7. MENOR VALOR COMPLETO
      // ===============================================

      return (
        a.costoTotal -
        b.costoTotal
      );

    }
  );

}

                STORAGE_KEYS.LISTA_COMPRA
              ),

              cargarDato<string>(
                STORAGE_KEYS.ULTIMA_COMPRA
              ),

              cargarDato<number>(
                STORAGE_KEYS.PRESUPUESTO
              ),
              cargarDato<unknown>(STORAGE_KEYS.PREPARACIONES),

            ]);


          const historialCargado = cargarHistorial(preparacionesGuardadas, hist);
          if (!activo) return;

          if (
            Array.isArray(ing)
          ) {
            setIngredientes(ing);
          }


          if (
            Array.isArray(cart)
          ) {
            setCarrito(cart);
          }


          setPreparaciones(historialCargado);


          if (
            Array.isArray(fav)
          ) {
            setFavoritos(fav);
          }


          if (
            Array.isArray(
              compradosGuardados
            )
          ) {
            setComprados(
              compradosGuardados
            );
          }


          if (
            Array.isArray(lista)
          ) {
            setListaCompra(lista);
          }


          if (
            typeof ultima ===
            'string'
          ) {
            setUltimaCompra(
              ultima
            );
          }


          if (
            typeof pres ===
              'number' &&
            Number.isFinite(pres)
          ) {

            setPresupuestoState(
              Math.max(
                0,
                pres
              )
            );

          }

          setHidratado(true);

        } catch (
          error
        ) {

          if (__DEV__) {

            console.warn(
              '⚠️ No se pudieron cargar los datos de Cocina en Crisis:',
              error
            );

          }

          if (activo) setErrorCarga(true);
        }

      };


    void cargar();
    return () => { activo = false; };

  }, [intentoCarga]);


  // ====================================================
  // 💾 GUARDAR DATOS
  // ====================================================

  useEffect(() => {

    /*
      No escribimos en almacenamiento
      hasta terminar la carga inicial.
    */

    if (!hidratado) {
      return;
    }

    let vigente = true;

    const guardar =
      async () => {

        try {

          await Promise.all([

            guardarDato(
              STORAGE_KEYS.INGREDIENTES,
              ingredientes
            ),

            guardarDato(
              STORAGE_KEYS.CARRITO,
              carrito
            ),

            guardarDato(
              STORAGE_KEYS.PREPARACIONES,
              preparaciones
            ),

            guardarDato(
              STORAGE_KEYS.FAVORITOS,
              favoritos
            ),

            guardarDato(
              STORAGE_KEYS.COMPRADOS,
              comprados
            ),

            guardarDato(
              STORAGE_KEYS.PRESUPUESTO,
              presupuesto
            ),

            guardarDato(
              STORAGE_KEYS.LISTA_COMPRA,
              listaCompra
            ),

            ...(ultimaCompra !== undefined
              ? [
                  guardarDato(
                    STORAGE_KEYS.ULTIMA_COMPRA,
                    ultimaCompra
                  ),
                ]
              : []),

          ]);

          if (vigente) setErrorGuardado(false);

        } catch (
          error
        ) {

              '⚠️ No se pudieron guardar los datos de Cocina en Crisis:',
              error
            );

          }
          if (vigente) setErrorGuardado(true);

        }

      };


    void guardar();
    return () => { vigente = false; };

  }, [
    hidratado,
    ingredientes,
    carrito,
    preparaciones,
    intentoGuardado,
    favoritos,
    comprados,
    presupuesto,
    listaCompra,
    ultimaCompra,
  ]);

  // ❤️ FAVORITOS
  // ====================================================

  const toggleFavorito =
    (
      id: string
    ) => {

      setFavoritos(
        prev => {

          if (
            prev.includes(id)
          ) {

            return prev.filter(
              item =>
                item !== id
            );

          }


          return [
            ...prev,
            id,
          ];

        }
      );

    };

  // ====================================================

  /*
    El presupuesto es un LÍMITE DE COMPRA.

    No representa una cuenta bancaria.

    Agregar algo a la despensa NO descuenta dinero.
    Cocinar tampoco descuenta dinero automáticamente.
  */

  const setPresupuesto =
    (
      monto: number
    ) => {

      if (
        !Number.isFinite(monto)
      ) {
        return;
      }


      setPresupuestoState(
        Math.max(
          0,
          monto

      );

  const agregarIngrediente =

          ?.trim()
          .toLowerCase();


      if (!ingrediente) {
        return;
      }


      setIngredientes(
        prev => {

          if (
            prev.includes(
              ingrediente
            )
          ) {

            return prev;

          }


          return [
            ...prev,
            ingrediente,
          ];

    (
      ingrediente: string

      const valor =
        ingrediente
          .trim()
          .toLowerCase();


      if (!valor) {
        return;
      }


      setIngredientes(
        prev =>
          prev.filter(
            item =>
              item !== valor
          )
      );

    };

                item !==
                ingrediente
            );

          }


          return [
            ...prev,
            ingrediente,
          ];

        }
      );

    };


  const limpiarComprados =
    () => {

      setComprados([]);

  // 🍳 CARRITO DE RECETAS

  const agregarAlCarrito =

      setComprados([]);

      );

  // ====================================================

  const marcarComprado =
    (
      ingrediente: string
    ) => {

      const valor =
        ingrediente
          .trim()
          .toLowerCase();


      if (!valor) {

      }

const nuevos =

            if (

// ==================================================

setComprados([]);

alert(

);

    };

  if (!hidratado) {

          </Pressable>
        </> : <ActivityIndicator color="#F2B35B" accessibilityLabel="Cargando tu cocina" />}
      </View>
    );
  }

  return (

    <IngredientesContext.Provider
      value={{
        ingredientes,
        ultimaCompra,

        favoritos,

        comprados,
        toggleComprado,

        presupuesto,
        setPresupuesto,

        carrito,

        historial,
        preparaciones,

        prepararRecetas,

      }}
    >

          <Text style={{ color: '#F2B35B', fontSize: 16 }}>Reintentar guardado</Text>
        </Pressable>
      </View>}
      {children}

    </IngredientesContext.Provider>

  );

}

    );

      ================================================= */}

                  ingredienteFaltante

                ingredienteFaltante

              Agrégalo a tu compra y desbloqueas esta receta.

                  </AppText>

                    item.id

      ================================================= */}

          </AppText>

          <AppText style={styles.sectionTitle}>

      ================================================= */}

        <AppText style={styles.finalTitle}>
          Ahora sí, vamos a cocinar

          }


          const datos =

  Number.isInteger(datos.paso) &&
  datos.paso > 0 &&

  datos.paso < receta.pasos.length

              />

            </View>


            <View style={styles.chefIntroIdentity}>

              <AppText style={styles.chefEyebrow}>
                CHEF LUNA
              </AppText>

              <AppText style={styles.chefIntroTitle}>
                Antes de comenzar
              </AppText>

            </View>


          </View>


          <AppText style={styles.chefIntroText}>
            {receta.chefInicio}
          </AppText>


        </View>

      )}


      {/* =================================================
          DISPONIBILIDAD PRINCIPAL
      ================================================= */}


      <View

          styles.readinessCard,
          listaParaCocinar &&
            styles.readinessCardReady,
        ]}
      >


        <View style={styles.readinessTop}>


          <View>

            <AppText style={styles.readinessEyebrow}>
              ESTADO DE TU DESPENSA
            </AppText>

            <AppText style={styles.readinessTitle}>

              {
                listaParaCocinar

                  ? 'Puedes cocinarla ahora'

                  : faltantes.length === 1

                    ? 'Estás a un ingrediente'

                    : `Te faltan ${faltantes.length} ingredientes`
              }

            </AppText>

          </View>


          <View
            style={[
              styles.readinessPercent,
              listaParaCocinar &&
                styles.readinessPercentReady,
            ]}
          >

            <AppText
              style={[
                styles.readinessPercentText,
                listaParaCocinar &&
                  styles.readinessPercentTextReady,
              ]}
            >
              {porcentajeDisponible}%
            </AppText>

          Tienes {ingredientesDisponibles} de {totalIngredientesClave} ingredientes clave.

      ================================================= */}

          subtitle="Chef Luna analizó tu despensa, costo y tiempo."

          ingredienteFaltante && (

              OPORTUNIDAD CHEF LUNA

                  ingredienteFaltante
                )?.nombre
                ??
                ingredienteFaltante
              }

              Agrégalo a tu compra y desbloqueas esta receta.
            </AppText>

                    {razon}
                  </AppText>

              <View style={styles.reasonDot} />

                obtenerIngrediente(

                    item.id
                  }

                  nombre={
                    info?.nombre
                    ??
                    item.id
                  }

                    `${item.cantidad} ${item.unidad}`
                  }

              );

            }
          )}

        </View>

      {/* =================================================
          FALTANTES
      ================================================= */}


      {faltantes.length > 0 && (

          <View style={styles.inventoryTop}>

            <AppText style={styles.inventoryEyebrow}>
              DISPONIBILIDAD
            </AppText>

            <AppText style={styles.inventoryPercent}>

            </AppText>

            faltantes

          NOTAS CHEF
      ================================================= */}


      <View style={styles.notesSection}>


        <View style={styles.notesHeader}>

          <AppText style={styles.sectionEyebrow}>
            CONOCIMIENTO CHEF LUNA
          </AppText>

          <AppText style={styles.sectionTitle}>
            Antes de encender la estufa
          </AppText>

          </AppText>

          <View style={styles.tipChef}>


            <View style={styles.tipHeader}>


              <View style={styles.tipIconChef}>

                <MaterialCommunityIcons
                  name="lightbulb-outline"

            </AppText>


          </View>

        {receta.ahorro && (

          <View style={styles.tipSave}>


            <View style={styles.tipHeader}>


              <View style={styles.tipIconSave}>

                <MaterialCommunityIcons
                  name="piggy-bank-outline"
                  size={21}
                  color="#9BC38A"
                />

              </View>


              <View>

                <AppText style={styles.tipEyebrowSave}>
                  AHORRO
                </AppText>

                <AppText style={styles.tipTitle}>
                  Cuida tu compra
                </AppText>

              </View>


            </View>


            <AppText style={styles.tipText}>
              {receta.ahorro}
            </AppText>


          </View>

        {receta.errorComun && (

          <View style={styles.tipWarning}>


            <View style={styles.tipHeader}>


              <View style={styles.tipIconWarning}>

                <AppText style={styles.tipEyebrowWarning}>
                  ERROR COMÚN
                </AppText>

                <AppText style={styles.tipTitle}>
                  Ojo con esto
                </AppText>

              </View>


            </View>


            <AppText style={styles.tipText}>
              {receta.errorComun}
            </AppText>


          </View>

        )}


      </View>

          CTA FINAL
      ================================================= */}


      <View style={styles.finalCard}>


        <View style={styles.finalIcon}>

          <MaterialCommunityIcons
            name="chef-hat"
            size={26}
            color="#F2B35B"
          />

        </View>


        <AppText style={styles.finalEyebrow}>
          TODO LISTO
        </AppText>


        <AppText style={styles.finalTitle}>
          Ahora sí, vamos a cocinar
        </AppText>


        <AppText style={styles.finalText}>
          Chef Luna te acompañará paso a paso durante la preparación.
        </AppText>

          activeOpacity={0.86}

       onPress={comenzarCocina}
>

            Empezar a cocinar

          <MaterialCommunityIcons
            name="arrow-right"
            size={19}

            return;
          }


          const datos =
            JSON.parse(
              guardado
            );


          if (
          
  typeof datos?.paso === 'number' &&
  Number.isInteger(datos.paso) &&
  datos.paso > 0 &&
  receta &&
  datos.paso < receta.pasos.length

    datos.paso

}

        } catch {

 }, [
